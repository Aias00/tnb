async function api(path, input, signal) {

}

async function perform(input, retry = false) {
  if (state.busy) { toast('正在处理上一条请求，请稍候'); throw new Error('正在处理上一条请求'); }
  if (state.pending && !retry) throw new Error('上一条请求结果尚未确认，请先重试原请求或放弃重试');
  const command = retry ? state.pending : { ...input, commandId: crypto.randomUUID() };
  if (!command) return;
  state.pending = command; save('pending', command); showPending(); state.busy = true; updateControls(); error('');
  try {
    const result = await api('api/command', command);
    state.pending = null; save('pending', null); showPending();
    if (command.action === 'submit') { setView('tasks'); state.selected = result.run.runId; save('selected', state.selected); $('submit-dialog').close(); const model = $('model').value; $('submit-form').reset(); $('model').value = model; save('model', model); save('draft', null); updateGrants(); }
    if (command.action === 'followup' && !command.resolveInputId) { $('followup').value = ''; save('followup:' + command.runId, null); }
    if (command.action === 'answer') save('answer:' + command.waitId + ':' + command.expectedVersion, null);
    if (command.action === 'answer' || command.resolveInputId) $('question-dialog').close();
    if (command.action === 'limits') $('limits-dialog').close();
    if (command.action === 'reconcile') $('reconcile-dialog').close();
    await refresh(); renderAttention(); toast(command.action === 'submit' ? '任务已持久保存，将在后台执行' : '操作已记录'); return result;
  } catch (issue) { error(issue.message); throw issue; }
  finally { state.busy = false; updateControls(); }
}

async function openChat(id) {
  if (state.chatPending && state.chatPending.chatId && state.chatPending.chatId !== id) { toast('请先确认上一条对话请求的结果'); return; }
  const sequence = ++state.chatOpenSequence; state.chatLoading = true;
  if (state.chatId !== id) { clearChatRecap(); resetChatPage(); state.chatSnapshot = null; $('chat-messages').replaceChildren(node('p','正在读取会话…','muted small')); $('chat-approvals').replaceChildren(); }
  state.chatStream?.close(); state.chatId = id; save('chat-selected', id); setView('chat'); updateChatControls();
  try { const snapshot = await api('api/chats/' + encodeURIComponent(id)); if (state.chatId !== id || sequence !== state.chatOpenSequence) return; renderChat(snapshot);
    const stream = new EventSource('api/chats/' + encodeURIComponent(id) + '/stream'); state.chatStream = stream;
    stream.onmessage = event => { if (state.chatId === id && sequence === state.chatOpenSequence) { try { renderChat(JSON.parse(event.data)); } catch (issue) { error(issue.message); } } };
    stream.onerror = () => { if (state.chatId === id && sequence === state.chatOpenSequence) { $('chat-status').textContent = '对话连接中断，正在重连；不会重新发送消息。'; api('api/info').catch(issue => error(issue.message)); } };
  } catch (issue) { if (sequence === state.chatOpenSequence) error(issue.message); }
  finally { if (sequence === state.chatOpenSequence) { state.chatLoading = false; updateChatControls(); } }
}
let pageController, pageSequence = 0;
function resetChatPage() { pageController?.abort(); pageSequence++; state.chatPage = null; state.chatPageBefore = null; state.chatPageTrail = []; state.chatPageLoading = false; }
async function loadChatPage(direction) {
  if (!state.chatId || state.chatPageLoading || !state.chatSnapshot) return;
  const chatId = state.chatId, displayed = state.chatPage || state.chatSnapshot;
  const before = direction === 'older' ? displayed.messages[0]?.id : state.chatPageTrail.at(-1);
  if (direction === 'newer' && before === null) { resetChatPage(); renderChat(state.chatSnapshot); return; }
  if (!before) return;
  const last = displayed.messages.at(-1)?.id, separator = last?.lastIndexOf(':');
  const previousBefore = state.chatPageBefore || (last && separator >= 0 ? last.slice(0,separator+1) + String(Number(last.slice(separator+1))+1).padStart(10,'0') : null);
  pageController?.abort(); const controller = pageController = new AbortController(), sequence = ++pageSequence;
  state.chatPageLoading = true; renderChat(state.chatSnapshot);
  try {
    const page = await api('api/chats/' + encodeURIComponent(chatId) + '/messages?before=' + encodeURIComponent(before), undefined, controller.signal);
    if (sequence !== pageSequence || controller.signal.aborted || chatId !== state.chatId) return;
    if (direction === 'older') { state.chatPageTrail.push(previousBefore); if (state.chatPageTrail.length > 256) state.chatPageTrail.shift(); }
    else state.chatPageTrail.pop();
    state.chatPage = page; state.chatPageBefore = before;
    renderChat(state.chatSnapshot); $('chat-messages').scrollTop = 0;
  } catch (issue) { if (sequence === pageSequence && !controller.signal.aborted) error(issue.message); }
  finally { if (sequence === pageSequence) { state.chatPageLoading = false; renderChat(state.chatSnapshot); } }
}
function renderChatStatus() {
  const snapshot = state.chatSnapshot; if (!snapshot) return;
  const chat = snapshot.chat, activity = snapshot.activity;
  const status = ({ idle:'本轮已完成，可以继续对话。', running:'正在执行', 'waiting-approval':'等待工具授权', interrupted:'本轮已中断，请核对工具结果后继续。', failed:'本轮执行失败，可核对后继续。' })[chat.status];
  let label = status;
  if (activity && ['running','waiting-approval'].includes(chat.status)) {
    label = ({ requesting:'正在请求模型', thinking:'正在思考', responding:'正在生成回复', 'preparing-tool':'正在准备工具', tool:'正在运行工具', retrying:'等待模型重试', 'waiting-approval':'等待工具授权', cancelling:'正在停止本轮' })[activity.phase] || status;
    if (activity.tool) label += ' · ' + activity.tool;
    if (activity.phase === 'retrying') label += (activity.retry ? ' · 第 ' + activity.retry + ' 次' : '') + (activity.retryAt ? ' · ' + Math.max(0,Math.ceil((activity.retryAt-Date.now())/1000)) + 's 后重试' : '');
    if (Number.isFinite(activity.startedAt)) label += ' · 本轮 ' + Math.max(0,Math.floor((Date.now()-activity.startedAt)/1000)) + 's';
  }
  $('chat-status').textContent = chat.error || label;
}

async function chatAction(input, retry = false) {
  if (state.chatBusy) throw new Error('正在处理对话请求');
  if (state.chatPending && !retry) throw new Error('请先重试原请求以确认结果');
  const command = retry ? state.chatPending : { ...input, commandId: crypto.randomUUID() }; if (!command) return;
  state.chatPending = command; save('chat-pending', command); state.chatBusy = true; updateChatControls(); error('');
  try {
    const result = await api('api/chat/command', command); state.chatPending = null; save('chat-pending', null);
    if (command.action === 'create') { state.chatId = result.id; save('chat-selected', result.id); state.chatSnapshot = null; }
    if (command.action === 'send') { resetChatPage(); $('chat-input').value = ''; save('chat-draft', null); }
    if (state.chatId) await openChat(state.chatId); return result;
  } catch (issue) { error(issue.message); throw issue; }
  finally { state.chatBusy = false; updateChatControls(); }
}
async function sendChat() {
  const text = $('chat-input').value.trim(); if (!text || state.chatBusy || state.chatPending) return;
  try { if (!state.chatId) { const [provider, model] = JSON.parse($('chat-model').value); await chatAction({ action: 'create', provider, model }); }
    await chatAction({ action: 'send', chatId: state.chatId, text });
  } catch (issue) { error(issue.message); }
}
