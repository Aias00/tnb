async function api(path, input) {
  const response = await fetch(path, { credentials: 'same-origin', ...(input === undefined ? {} : { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(input) }) });
  const value = await response.json();
  if (!response.ok) { if (response.status === 401) { state.stream?.close(); state.chatStream?.close(); connection(false, '授权已失效'); } throw new Error(value.error || '请求失败'); }
  return value;
}

function showPending() { $('retry-box').classList.toggle('hidden', !state.pending); $('submit-form').querySelector('[type=submit]').textContent = state.pending?.action === 'submit' ? '重试原提交' : '提交任务'; }
async function perform(input, retry = false) {
  if (state.busy) { toast('正在处理上一条请求，请稍候'); throw new Error('正在处理上一条请求'); }
  if (state.pending && !retry) throw new Error('上一条请求结果尚未确认，请先重试原请求或放弃重试');
  const command = retry ? state.pending : { ...input, commandId: crypto.randomUUID() };
  if (!command) return;
  state.pending = command; save('pending', command); showPending(); state.busy = true; updateControls(); error('');
  try {
    const result = await api('api/command', command);
    state.pending = null; save('pending', null); showPending();
    if (command.action === 'submit') { setView('tasks'); state.selected = result.run.runId; save('selected', state.selected); $('submit-dialog').close(); const model = $('model').value; $('submit-form').reset(); $('model').value = model; save('model', model); save('draft', null); updateGrants(); }
    if (command.action === 'followup' && !command.resolveInputId) { $('followup').value = ''; save('followup:' + command.runId, null); }
    if (command.action === 'answer') save('answer:' + command.waitId + ':' + command.expectedVersion, null);
    if (command.action === 'answer' || command.resolveInputId) $('question-dialog').close();
    if (command.action === 'limits') $('limits-dialog').close();
    if (command.action === 'reconcile') $('reconcile-dialog').close();
    await refresh(); renderAttention(); toast(command.action === 'submit' ? '任务已持久保存，将在后台执行' : '操作已记录'); return result;
  } catch (issue) { error(issue.message); throw issue; }
  finally { state.busy = false; updateControls(); }
}

async function loadDetail() {

  const selected = state.selected, sequence = ++state.sequence;

    if (sequence !== state.sequence || selected !== state.selected) return;

  } catch (issue) { error(issue.message); }

async function openChat(id) {
  if (state.chatPending && state.chatPending.chatId && state.chatPending.chatId !== id) { toast('请先确认上一条对话请求的结果'); return; }
  state.chatStream?.close(); state.chatId = id; save('chat-selected', id); setView('chat');
  try { const snapshot = await api('api/chats/' + encodeURIComponent(id)); if (state.chatId !== id) return; renderChat(snapshot);
    const stream = new EventSource('api/chats/' + encodeURIComponent(id) + '/stream'); state.chatStream = stream;
    stream.onmessage = event => { if (state.chatId === id) { try { renderChat(JSON.parse(event.data)); } catch (issue) { error(issue.message); } } };
    stream.onerror = () => { if (state.chatId === id) { $('chat-status').textContent = '对话连接中断，正在重连；不会重新发送消息。'; api('api/info').catch(issue => error(issue.message)); } };
  } catch (issue) { error(issue.message); }
}

async function chatAction(input, retry = false) {
  if (state.chatBusy) throw new Error('正在处理对话请求');
  if (state.chatPending && !retry) throw new Error('请先重试原请求以确认结果');
  const command = retry ? state.chatPending : { ...input, commandId: crypto.randomUUID() }; if (!command) return;
  state.chatPending = command; save('chat-pending', command); state.chatBusy = true; updateChatControls(); error('');
  try {
    const result = await api('api/chat/command', command); state.chatPending = null; save('chat-pending', null);
    if (command.action === 'create') { state.chatId = result.id; save('chat-selected', result.id); state.chatSnapshot = null; }
    if (command.action === 'send') { $('chat-input').value = ''; save('chat-draft', null); }
    if (state.chatId) await openChat(state.chatId); return result;
  } catch (issue) { error(issue.message); throw issue; }
  finally { state.chatBusy = false; updateChatControls(); }
}
async function sendChat() {
  const text = $('chat-input').value.trim(); if (!text || state.chatBusy || state.chatPending) return;
  try { if (!state.chatId) { const [provider, model] = JSON.parse($('chat-model').value); await chatAction({ action: 'create', provider, model }); }
    await chatAction({ action: 'send', chatId: state.chatId, text });
  } catch (issue) { error(issue.message); }
}
